{< BV号 播放开始时间 播放自动开始 分P 是否静音 >}
此为变量顺序，类python方法赋值，不写明则为默认值，不可乱序赋值。
例：

  {< bilibili BV11i4y1b7pd>}
  {< bilibili BV11i4y1b7pd "1:14">}   1:14开始播放，默认不自动播放
  {< bilibili BV11i4y1b7pd "1:14" 1 >}  1:14开始播放，自动播放
  {< bilibili BV11i4y1b7pd "1:14" 1 1 0>} 1:14开始播放，自动播放视频中的1p，不静音 -->
但是浏览器总会禁止自动播放的视频的音量，目前静音算是无效。

    function showHideView() {
        // 判断是否存在vercount标签
        let viewCounts = document.querySelectorAll("#viewCount");
        if (viewCounts) {
            // 判断是否为文章页

// 分类颜色分配系统
// 为每个分类名称分配固定颜色，确保相同分类在任何地方显示相同颜色

interface CategoryColorMap {
    [key: string]: string;
}

class CategoryColorManager {
    private static instance: CategoryColorManager;
    private colorMap: CategoryColorMap = {};
    private readonly colors: string[] = [
        '#8ea885', // 绿色系
        '#df7988', // 粉色系
        '#0177b8', // 蓝色系
        '#ffb900', // 黄色系
        '#6b69d6', // 紫色系
        '#e86c60', // 红色系
        '#1abc9c', // 青色系
        '#9b59b6', // 紫红色系
        '#3498db', // 天蓝色系
        '#e67e22', // 橙色系
        '#2ecc71', // 翡翠绿
        '#e74c3c', // 红色
        '#f1c40f', // 黄色
        '#16a085', // 海绿色
        '#d35400', // 南瓜色
        '#7f8c8d', // 石灰色
        '#2c3e50', // 湿沥青色
        '#c0392b', // 猩红色
        '#27ae60', // 绿松石色
        '#8e44ad'  // 木槿色
    ];
    private readonly textColor: string = '#ffffff';

    private constructor() {}

    public static getInstance(): CategoryColorManager {
        if (!CategoryColorManager.instance) {
            CategoryColorManager.instance = new CategoryColorManager();
        }
        return CategoryColorManager.instance;
    }

    // 简单的哈希函数，将字符串转换为数字
    private hashCode(str: string): number {
        let hash = 0;
        for (let i = 0; i < str.length; i++) {
            const char = str.charCodeAt(i);
            hash = ((hash << 5) - hash) + char;
            hash = hash & hash; // 转换为32位整数
        }
        return Math.abs(hash);
    }

    // 动态生成颜色函数
    private generateColor(name: string): string {
        const hash = this.hashCode(name);
        // 为了更好的视觉效果，我们调整参数范围
        const hue = hash % 360;
        const saturation = 65 + (hash % 25); // 65-90%
        const lightness = 45 + (hash % 15);  // 45-60%
        
        return `hsl(${hue}, ${saturation}%, ${lightness}%)`;
    }

    // 获取分类颜色
    public getCategoryColor(categoryName: string): string {
        // 如果已经有这个分类的颜色，直接返回
        if (this.colorMap[categoryName]) {
            return this.colorMap[categoryName];
        }

        // 获取分类索引（基于已分配颜色的数量）
        const index = Object.keys(this.colorMap).length;
        
        let color: string;
        // 如果索引在预定义颜色范围内，使用预定义颜色
        if (index < this.colors.length) {
            color = this.colors[index];
        } else {
            // 否则动态生成颜色
            color = this.generateColor(categoryName);
        }
        
        this.colorMap[categoryName] = color;
        return color;
    }

    // 应用颜色到所有分类标签
    public applyCategoryColors(): void {
        // 查找所有分类标签
        // const categoryElements = document.querySelectorAll('.article-category a, .tagCloud-tags a, .widget-categories a');
        const categoryElements = document.querySelectorAll('.article-category a');
        
        // 遍历所有分类标签
        categoryElements.forEach(element => {
            const link = element as HTMLAnchorElement;
            // 获取分类名称（链接文本）
            const categoryName = link.textContent?.trim();
            
            if (categoryName) {
                // 获取并应用颜色
                const backgroundColor = this.getCategoryColor(categoryName);
                
                // 应用样式
                link.style.backgroundColor = backgroundColor;
                link.style.color = this.textColor;
                
                // 确保样式优先级
                link.style.setProperty('background-color', backgroundColor, 'important');
                link.style.setProperty('color', this.textColor, 'important');
            }
        });
    }
}

// 初始化分类颜色管理器
document.addEventListener('DOMContentLoaded', () => {
    const colorManager = CategoryColorManager.getInstance();
    colorManager.applyCategoryColors();
    
    // 监听可能的页面更新（例如通过平滑导航加载新内容）
    const observer = new MutationObserver(() => {
        colorManager.applyCategoryColors();
    });
    
    observer.observe(document.body, {
        childList: true,
        subtree: true
    });
});

// 也监听页面加载事件
window.addEventListener('load', () => {
    const colorManager = CategoryColorManager.getInstance();
    colorManager.applyCategoryColors();
});

// 导出以便其他模块使用
export default CategoryColorManager;
